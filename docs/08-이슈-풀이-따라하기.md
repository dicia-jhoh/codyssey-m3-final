## 약속 세 가지

1. **한 번에 하나, 적힌 순서대로.** 앞 PR 이 머지된 뒤에 다음 걸 해요. 같은 파일을 동시에 고치면 서로 엉켜요.
2. **머지 버튼은 누르지 않아요.** 오주현이 확인하고 머지해요. 테스트·화면 확인·스크린샷도 오주현이 해요.
3. **👤 표시 칸은 지어내지 않아요.** 사람이 직접 해 보고 본 것만 적어요. 못 했으면 「못 함」이라고 적어요.

### #255 개인정보 가림 — 전화번호가 「계좌번호」로 적히는 문제

> 고지서의 `010-1234-5678`·`042-123-4567` 이 화면에 「[계좌번호 가림]」으로 나와요. 가리는 건 맞는데 이름표가 틀려서 어르신이 헷갈려요.

찾을 글

```python
def redact(text: str) -> str:
    """주민번호는 앞 6자리와 성별 숫자만 남기고, 계좌번호는 통째로 가린다."""

    return _ACCOUNT.sub(_hide_account, text)
```

바꿀 글

```python
# 전화번호: 0 으로 시작하는 지역·휴대폰 번호 - 3~4자리 - 4자리. 계좌번호와 모양이 비슷해서
# 계좌보다 먼저 가린다 — 안 그러면 「[계좌번호 가림]」으로 잘못 적힌다
_PHONE = re.compile(r"(?<![\d-])0\d{1,2}-\d{3,4}-\d{4}(?![\d-])")


def redact(text: str) -> str:
    """주민번호는 앞 6자리와 성별 숫자만 남기고, 전화번호·계좌번호는 통째로 가린다."""
    text = _RRN.sub(lambda m: f"{m.group(1)}-{m.group(2)}******", text)
    text = _PHONE.sub("[전화번호 가림]", text)
    return _ACCOUNT.sub(_hide_account, text)
```

### #59 조문 번호 → JO 코드 변환 함수

### #60 원문 대조 함수

> 두 글을 겹쳐 보고 「띄어쓰기만 다르고 글자는 똑같나?」를 봐요. 한 글자라도 다르면 「아니요」예요.

나간다(#32). 사람 눈 대조 대신 법제처 원문과 기계로 맞춰 본다.
"""

from __future__ import annotations

import re


def jo_code(article_no: str) -> str:
    """조문 번호("80", "80의2")를 법제처 API 의 6자리 JO 코드로 바꾼다.

    조 번호 4자리(앞을 0으로 채움) + 「의N」 2자리(없으면 00).
    "80" → "008000", "80의2" → "008002", "234의2" → "023402".
    """
    m = re.fullmatch(r"(\d+)(?:의(\d+))?", article_no.strip())
    if not m:
        raise ValueError(f"조문 번호 모양이 아니다: {article_no!r} (예: \"80\", \"80의2\")")
    return f"{int(m.group(1)):04d}{int(m.group(2) or 0):02d}"


def _squash(text: str) -> str:
    """공백·줄바꿈을 전부 지운다 — 원문과 quote 의 띄어쓰기 차이(「말한다) 까지」 vs 「말한다)까지」)를 무시하려고."""
    return re.sub(r"\s+", "", text)


def quote_matches(article_text: str, quote: str) -> bool:
    """quote 가 조문 원문 안에 글자 그대로 들어 있는가(공백 차이만 무시).

    """
    q = _squash(quote)
    return bool(q) and q in _squash(article_text)

### #61 법제처 API 호출 함수 2개

> 법 이름으로 법의 번호(MST)를 찾고, 그 번호와 조문 코드로 원문을 받아 와요. 테스트는 진짜 서버 대신 가짜 대답으로 해요.

이슈: [#61](https://github.com/dicia-jhoh/codyssey-m3-final/issues/61) · 브랜치 이름: `gihoon/61-law-api`

**[`backend/app/lawapi.py`](https://github.com/dicia-jhoh/codyssey-m3-final/edit/main/backend/app/lawapi.py)** — `Ctrl+A` 로 전부 고른 뒤 아래를 통째로 붙여요(원래 글은 지워져요).

```python
"""법제처 Open API 로 조문 원문을 받아 articles.jsonl 의 quote 와 대조하는 도구.

articles.jsonl 의 quote 가 원문과 한 글자라도 다르면 화면에 "원문 인용"으로 틀린 글이
나간다(#32). 사람 눈 대조 대신 법제처 원문과 기계로 맞춰 본다.
"""

from __future__ import annotations

import re

import httpx

_BASE = "https://www.law.go.kr/DRF"


def jo_code(article_no: str) -> str:
    """조문 번호("80", "80의2")를 법제처 API 의 6자리 JO 코드로 바꾼다.

    조 번호 4자리(앞을 0으로 채움) + 「의N」 2자리(없으면 00).
    "80" → "008000", "80의2" → "008002", "234의2" → "023402".
    """
    m = re.fullmatch(r"(\d+)(?:의(\d+))?", article_no.strip())
    if not m:
        raise ValueError(f"조문 번호 모양이 아니다: {article_no!r} (예: \"80\", \"80의2\")")
    return f"{int(m.group(1)):04d}{int(m.group(2) or 0):02d}"


def _squash(text: str) -> str:
    """공백·줄바꿈을 전부 지운다 — 원문과 quote 의 띄어쓰기 차이(「말한다) 까지」 vs 「말한다)까지」)를 무시하려고."""
    return re.sub(r"\s+", "", text)


def quote_matches(article_text: str, quote: str) -> bool:
    """quote 가 조문 원문 안에 글자 그대로 들어 있는가(공백 차이만 무시).

    quote 는 조문의 일부(①항 하나, 3호 하나)라서 「같다」가 아니라 「포함된다」로 본다.
    """
    q = _squash(quote)
    return bool(q) and q in _squash(article_text)


def _as_list(value) -> list:
    """법제처 JSON 은 항목이 하나면 리스트 대신 그 항목 하나만 준다 — 늘 리스트로 맞춘다."""
    if value is None:
        return []
    return value if isinstance(value, list) else [value]


def find_mst(law_name: str, oc: str) -> str:
    """법령 이름이 정확히 같은 현행 법령의 법령일련번호(MST)를 찾는다."""
    r = httpx.get(
        f"{_BASE}/lawSearch.do",
        params={"OC": oc, "target": "law", "type": "JSON", "query": law_name},
        timeout=20,
    )
    r.raise_for_status()
    for law in _as_list(r.json().get("LawSearch", {}).get("law")):
        if law.get("법령명한글") == law_name and law.get("현행연혁코드") == "현행":
            return law["법령일련번호"]
    raise LookupError(f"현행 법령을 못 찾았다: {law_name}")


def fetch_article(mst: str, jo: str, oc: str) -> str:
    """MST + JO 코드로 조문 하나를 받아 본문 텍스트(조문·항·호·목)를 줄바꿈으로 이어 돌려준다."""
    r = httpx.get(
        f"{_BASE}/lawService.do",
        params={"OC": oc, "target": "law", "type": "JSON", "MST": mst, "JO": jo},
        timeout=20,
    )
    r.raise_for_status()
    units = _as_list((r.json().get("법령", {}).get("조문") or {}).get("조문단위"))
    lines: list[str] = []
    for unit in units:
        lines.append(unit.get("조문내용", ""))
        for hang in _as_list(unit.get("항")):
            lines.append(hang.get("항내용", ""))
            for ho in _as_list(hang.get("호")):
                lines.append(ho.get("호내용", ""))
                for mok in _as_list(ho.get("목")):
                    lines.append(mok.get("목내용", ""))
    text = "\n".join(line for line in lines if line)
    if not text:
        raise LookupError(f"조문을 못 찾았다: MST={mst} JO={jo}")
    return text
```

LAW_OC 환경변수(없으면 팀 공용 parapuda). 조문을 추가하는 절차는 data/legal/README.md.

from __future__ import annotations

import json
import os
import sys
from pathlib import Path

from app import lawapi
from app.rag import _format_law  # 화면 근거와 같은 「제80조의2」 표기

_DEFAULT_ARTICLES = Path(__file__).resolve().parents[2] / "data" / "legal" / "articles.jsonl"


def verify(path: Path, oc: str) -> int:
    """조문마다 결과 한 줄을 찍고 FAIL 개수를 돌려준다."""
    failed = 0
    for line in path.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        row = json.loads(line)
        name = _format_law(row["law_name"], row["article_no"])
        try:
            mst = lawapi.find_mst(row["law_name"], oc)
            text = lawapi.fetch_article(mst, lawapi.jo_code(row["article_no"]), oc)
        except (LookupError, ValueError) as e:
            print(f"[FAIL] {name} — {e}")
            failed += 1
            continue
        if lawapi.quote_matches(text, row["quote"]):
            print(f"[OK] {name}")
        else:
            print(f"[FAIL] {name} — 원문: {text}")
            failed += 1
    return failed


def main(argv: list[str] | None = None) -> int:
    path = Path(argv[0]) if argv else _DEFAULT_ARTICLES
    if not path.exists():
        print(f"[!] 파일 없음: {path}")
        return 1
    return 1 if verify(path, os.environ.get("LAW_OC", "parapuda")) else 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv[1:]))

### #99 테스트셋 첫 5장 정답 적기 👤

> 가족 고지서 5장을 찍어서(이름·주소·계좌는 손으로 가리고) 「얼마·언제까지·뭘 챙기나」 정답을 한 줄씩 적어요. 사진은 팀 드라이브에만 올려요.

이슈: [#99](https://github.com/dicia-jhoh/codyssey-m3-final/issues/99) · 브랜치 이름: `gihoon/99-testset-5`

**[`data/testset/labels.jsonl`](https://github.com/dicia-jhoh/codyssey-m3-final/new/main/data/testset?filename=labels.jsonl)** — 새 파일이에요. 아래 5줄을 붙이고 👤 를 사진 보고 채워요.

```json
{"image": "images/001.jpg", "doc_type": "고지서", "money": [👤48200], "deadlines": ["👤2026-10-31"], "documents": [], "todos": [{"due": "👤2026-10-31", "amount": 👤48200}]}
{"image": "images/002.jpg", "doc_type": "고지서", "money": [👤], "deadlines": ["👤"], "documents": [], "todos": [{"due": "👤", "amount": 👤}]}
{"image": "images/003.jpg", "doc_type": "고지서", "money": [👤], "deadlines": ["👤"], "documents": [], "todos": [{"due": "👤", "amount": 👤}]}
{"image": "images/004.jpg", "doc_type": "가정통신문", "money": [], "deadlines": ["👤"], "documents": ["👤"], "todos": [{"due": "👤", "amount": null}]}
{"image": "images/005.jpg", "doc_type": "안내문", "money": [], "deadlines": ["👤"], "documents": ["👤"], "todos": [{"due": "👤", "amount": null}]}
```

- 고지서 3장 이상 + 가정통신문·안내문 1장 이상을 섞어요. 문서 종류 이름은 `고지서`·`가정통신문`·`안내문`·`신청서` 중에서 골라요.
- 금액은 쉼표 없이 숫자만(`48200`), 날짜는 `2026-10-31` 모양. 없는 건 `[]`(빈 칸), 돈 없는 할 일은 `"amount": null`.
- 👤 글자는 하나도 남기지 않아요. 따옴표·쉼표·괄호를 지우면 파일이 깨져요 — 오주현이 리뷰 때 검사해요.
- 이름·주소·주민번호·계좌는 정답에도 적지 않아요.
